SELECT s.sname, s.city, o.odate, o.onum, o.amt
FROM Salespeople s
JOIN (
    SELECT o.*,
           RANK() OVER (
               PARTITION BY snum
               ORDER BY amt DESC
           ) AS order_rank
    FROM Orders o
) o ON s.snum = o.snum
WHERE o.order_rank = 1;
