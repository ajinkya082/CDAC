Ans --->
create table sales_order1(
s_order_no varchar(6) primary key check(s_order_no like('o%')),
s_order_date date,
client_no varchar(6) ,foreign key(client_no) references client_master1(client_no),
dely_addr varchar(25),
salesman_no varchar(6) ,foreign key(salesman_no) references salesman_master1(salesman_no),
dely_type char(1) check(dely_type in('P','F')) default 'F',
billed_yn char(1) check(billed_yn in('Y','N')) default 'N',
dely_date date,
order_status varchar(10) check(order_status in('in process','Fulfilled','BackOrder','Canceled')));
