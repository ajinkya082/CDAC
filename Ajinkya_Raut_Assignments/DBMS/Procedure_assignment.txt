	create procedure ass_eg(in pename varchar(10))

3.create a procedure "ass_eg3" to accept a deptno and mgr from the user and display all the details for employees in the same deptno and whose manager is the given value

delimiter //
create procedure ass_eg3(in pdeptno numeric(2),in pmgr numeric(4))
begin
select * from emp where deptno=pdeptno and mgr=pmgr; 
end
