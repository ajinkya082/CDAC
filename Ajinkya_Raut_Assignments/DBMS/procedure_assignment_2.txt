>= 60 display first class , 
<60 and >=50 display 2nd class
>=35 and  <50 : pass class
< 35 failed

secon
 delimiter //
 create procedure check_marks(in pmarks int )
   begin
     declare result varchar(20);
     set @mark=pmarks;
     if @mark>=60 then
     set result='first class';
     elseif @mark<60 and @mark>=50 then
     set result ='second class';
     elseif @mark<50 and @mark>=35 then
     set result ='pass class';
     elseif @mark<35 then
     set result = 'failed';
     end if;
     select result;
     end
     //

5. calculate the total of first 20 number using while loop
delimiter //
 create procedure totalsum()
     begin
     set @number=1;
     set @sum=0;
     while(@number<=20)
     do
     select @sum;
     set @sum=@sum+@number;
     set @number=@number+1;
     end while;
     end

4. display the first 10 even numbers
delimiter //
 create procedure evenno()
     begin
     declare result varchar(20);
     set @number=1;
     while(@number<=20)
     do
     if mod(@number,2)=0 then
     set result='even number';
     select @number;
     end if;
     set @number=@number+1;
     end while;
     end 
     //

7. accept a deptno from the user and return number of employees working in the same department and the max salary for  the same dept
delimiter //

create procedure dept_details(in pdeptno int)
begin
    select count(*) as no_of_employees, max(sal) as max_salary
    from emp
    where deptno = pdeptno;
end //

create procedure sal_detail(in pdeptno int)
begin
    select max(sal) as total_salary, max(sal) as max_salary,min(sal) as min_salary
    from emp
    where deptno = pdeptno;
end //
